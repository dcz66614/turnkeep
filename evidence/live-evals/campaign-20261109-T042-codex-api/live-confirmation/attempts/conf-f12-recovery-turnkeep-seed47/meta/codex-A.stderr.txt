def taint_descendants(db, scope_id, root_ids, changes):

def insert_record(db, scope_id, key, encoded, kind, sources, parents, changes,
                  *, state="active", replacements=()):

    validate_parents(db, scope_id, parents, child_id=record_id)
    db.execute("INSERT INTO memories(id, scope_id, key, value_json, kind, state, revision, created_at, updated_at) "
               "VALUES (?,?,?,?,?,?,?,?,?)",
               (record_id, scope_id, key, encoded, kind, state, 1, now, now))
    link_sources(db, scope_id, record_id, sources)
    for parent in parents:
