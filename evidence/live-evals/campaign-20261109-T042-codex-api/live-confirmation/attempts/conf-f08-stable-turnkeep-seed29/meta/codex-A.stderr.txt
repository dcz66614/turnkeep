def _changed_since(db, scope_id, run):

    now = db.execute("SELECT id, key, state, revision FROM memories WHERE scope_id=?",
                     (scope_id,)).fetchall()

    now_map = {r["id"]: r for r in now}

        if old is None or old["revision"] != r["revision"] or old["state"] != r["state"]:
